### Task B — Extract `LPStatementService`

**Why second:** Eliminates the three-way duplication before adding the Catalog screen (which would have needed a fourth copy).

**File to create:** `30-LoyaltyProgram/LPStatementService.swift`

```swift
import Foundation

// MARK: - LPStatementService
// Stateless service: encode/decode LPPointTransaction arrays from AppStorage Data.
enum LPStatementService {
    /// Prepends `tx` to the existing transaction list stored in `data`.
    /// Silent on codec failure — logs via ZodiakLog.
    static func appendTransaction(_ tx: LPPointTransaction, to data: inout Data) {
        var existing: [LPPointTransaction] = decodeTransactions(from: data)
        existing.insert(tx, at: 0)
        do {
            data = try JSONEncoder().encode(existing)
        } catch {
            ZodiakLog.warning(.service, "LP statement encode failed",
                              metadata: ["feature": "LoyaltyProgram",
                                         "exception.type": String(describing: type(of: error))])
        }
    }

    /// Decodes the transaction list from `data`. Returns `[]` for empty or corrupt data.
    static func decodeTransactions(from data: Data) -> [LPPointTransaction] {
        guard !data.isEmpty else { return [] }
        do {
            return try JSONDecoder().decode([LPPointTransaction].self, from: data)
        } catch {
            ZodiakLog.warning(.service, "LP statement decode failed",
                              metadata: ["feature": "LoyaltyProgram",
                                         "exception.type": String(describing: type(of: error))])
            return []
        }
    }
}
```

### Task D — Add CPF checksum validation to `ValidationService`

**Why fourth:** Security improvement; no visual impact; isolated.

**File to modify:** `ZodiakiOS/ZodiakiOS/Services/Services.swift`

Add to `enum ValidationService`:
```swift
/// Validates a Brazilian CPF (Cadastro de Pessoas Físicas) using the
/// official Receita Federal two-digit checksum algorithm.
/// - Parameter digits: Exactly 11 numeric characters (no formatting).
/// - Returns: `true` when length is 11, not all-same-digit, and both check digits are valid.
static func validateCPF(_ digits: String) -> Bool {
    guard digits.count == 11, digits.allSatisfy(\.isNumber) else { return false }
    // Reject trivial sequences (00000000000, 11111111111, …)
    guard Set(digits).count > 1 else { return false }

    func checkDigit(for slice: Substring, multiplierStart: Int) -> Character {
        let sum = zip(slice, stride(from: multiplierStart, through: 2, by: -1))
            .reduce(0) { acc, pair in acc + Int(String(pair.0))! * pair.1 }
        let remainder = (sum * 10) % 11
        return remainder == 10 ? "0" : Character(String(remainder))
    }

    let d10 = checkDigit(for: digits.prefix(9), multiplierStart: 10)
    guard digits[digits.index(digits.startIndex, offsetBy: 9)] == d10 else { return false }
    let d11 = checkDigit(for: digits.prefix(10), multiplierStart: 11)
    return digits[digits.index(digits.startIndex, offsetBy: 10)] == d11
}
```
